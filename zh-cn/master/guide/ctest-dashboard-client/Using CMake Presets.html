Step 8: Using CMake Presets¶

Added in version 4.4.

This step extends Step 3 by moving the coverage settings into a preset, instead of passing them with the OPTIONS argument of ctest_configure().

Step8/CMakePresets.json¶
{
  "version": 10,
  "cmakeMinimumRequired": {
    "major": 4,
    "minor": 4,
    "patch": 0
  },
  "configurePresets": [
    {
      "name": "coverage",
      "displayName": "Coverage",
      "description": "Build with gcov coverage instrumentation",
      "generator": "Unix Makefiles",
      "cacheVariables": {
        "CMAKE_BUILD_TYPE": "Debug",
        "CMAKE_CXX_FLAGS": "--coverage"
      }
    }
  ],
  "buildPresets": [
    {
      "name": "coverage",
      "configurePreset": "coverage"
    }
  ],
  "testPresets": [
    {
      "name": "coverage",
      "configurePreset": "coverage"
    }
  ]
}
Step8/submit.cmake¶
cmake_minimum_required(VERSION 4.4)

set(CTEST_BUILD_NAME "Step8: presets")
site_name(CTEST_SITE)
set(CTEST_SOURCE_DIRECTORY "${CTEST_SCRIPT_DIRECTORY}/../Project")
set(CTEST_BINARY_DIRECTORY "${CTEST_SCRIPT_DIRECTORY}/build")

# Note that we aren't setting CTEST_CMAKE_GENERATOR in this script any more.
# It is now defined in the configure preset instead.

set(CTEST_NOTES_FILES "${CMAKE_CURRENT_LIST_FILE}")

set(CTEST_UPDATE_COMMAND "git")
set(CTEST_UPDATE_VERSION_ONLY 1)

set(CTEST_USE_LAUNCHERS 1)
set(CTEST_COVERAGE_COMMAND "gcov")

# Select the preset, and the presets file that defines it, for the
# ctest_configure(), ctest_build(), and ctest_test() commands below.
set(CTEST_PRESET "coverage")
set(CTEST_PRESETS_FILE "${CTEST_SCRIPT_DIRECTORY}/CMakePresets.json")

ctest_empty_binary_directory("${CTEST_BINARY_DIRECTORY}")

ctest_start(Experimental)
ctest_update()
ctest_configure()
ctest_build()
ctest_test()
ctest_coverage()
ctest_submit()

Setting CTEST_PRESET selects a preset with that name for the ctest_configure(), ctest_build(), and ctest_test() commands. The commands read the variable themselves, so they need no arguments. Setting CTEST_PRESETS_FILE reads the presets from a file other than the CMakePresets.json file in the source tree. Here, the presets file is in the Step8 directory.

Presets keep the build settings in a standard file, instead of in the dashboard script. Other scripts, cmake --preset, and IDEs can use the same presets.

Requirements¶

  • GCC or Clang with --coverage.

  • A gcov tool that matches the compiler version, found in the PATH.

Running the Script¶

ctest -S Step8/submit.cmake -V
