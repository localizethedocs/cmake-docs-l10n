Step 3: Collecting Coverage¶

This step extends Step 2 by collecting code coverage and submitting it to CDash. It does not enable the build error from Step 2.

Step3/submit.cmake¶
set(CTEST_BUILD_NAME "Step3: coverage")
site_name(CTEST_SITE)
set(CTEST_SOURCE_DIRECTORY "${CTEST_SCRIPT_DIRECTORY}/../Project")
set(CTEST_BINARY_DIRECTORY "${CTEST_SCRIPT_DIRECTORY}/build")
set(CTEST_CMAKE_GENERATOR "Unix Makefiles")
set(CTEST_NOTES_FILES "${CMAKE_CURRENT_LIST_FILE}")

set(CTEST_UPDATE_COMMAND "git")
set(CTEST_UPDATE_VERSION_ONLY 1)

set(CTEST_USE_LAUNCHERS 1)

# Tell CTest to use gcov for coverage analysis.
set(CTEST_COVERAGE_COMMAND "gcov")

ctest_empty_binary_directory("${CTEST_BINARY_DIRECTORY}")

ctest_start(Experimental)
ctest_update()

# Build with the --coverage flag to enable gcov-style instrumentation.
set(options
  "-DCMAKE_BUILD_TYPE=Debug"
  "-DCMAKE_CXX_FLAGS=--coverage"
  )
ctest_configure(OPTIONS "${options}")

ctest_build()
ctest_test()

# Run gcov to find which lines of code the tests executed.
ctest_coverage()

ctest_submit()

Two things are needed to collect coverage:

  1. Build the project with instrumentation. The --coverage flag makes GCC and Clang add counters that record which lines run. A Debug build avoids optimizations that would make the line counts less accurate.

  2. Run the coverage tool. After ctest_test(), the ctest_coverage() command runs the tool given by CTEST_COVERAGE_COMMAND on the data the tests produced. CTest parses its output and submits the coverage of each file.

On CDash, the build shows the overall coverage percentage. The details show the coverage of each file, with the lines that ran and the lines that did not. For example, the fails test never runs one line of fails.cxx:

Project/fails.cxx¶
#include <iostream>

int main()
{
  bool b = false;
  if (b) {
    // This line never runs, so coverage reports it as missed.
    std::cout << "this shouldn't happen..." << std::endl;
  }

  std::cout << "this test fails" << std::endl;
  return 1;
}

Requirements¶

  • GCC or Clang (supports --coverage / -fprofile-arcs -ftest-coverage)

  • A gcov tool that matches the compiler version, found in the PATH.

See the CTest Coverage Step for other supported tools.

Running the Script¶

ctest -S Step3/submit.cmake -V
